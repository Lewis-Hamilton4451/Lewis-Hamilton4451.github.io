    <?php
    # flag in config.php
    include("config.php");
    if(isset($_GET['c'])){
            $c = $_GET['c'];
            if(!preg_match("/system|exec|highlight|cat|\(|\.|\;|file|php|config/i",$c)){
                    eval($c);
            }else{
                die("cmd error");
            }
    }else{
            highlight_file(__FILE__);
    }
    ?>
